MODEL OPTIMIZATION PIPELINE:
────────────────────────────
# Install tools
npm install -g gltf-transform
npm install -g gltfpack

# Step 1: Draco compress
gltf-transform draco input.glb output.glb \
  --method edgebreaker \
  --quantize-position 14 \
  --quantize-normal 10 \
  --quantize-texcoord 12

# Step 2: Alternative meshopt compression
gltfpack -i input.glb -o output.glb -cc -tc -noq

# Step 3: Verify
# Load in: https://gltf-viewer.donmccurdy.com/
# Check: polygons < 80k, file size < 5MB

# Step 4: Create LOD variants
# LOD0: Full quality (desktop)     → ~80k polys → car-draco.glb
# LOD1: Medium (tablet)            → ~40k polys → car-lod1.glb
# LOD2: Low (mobile)               → ~15k polys → car-lod2.glb
# LOD3: Image fallback (low-power) → Static render PNG

function HeroCamera() {
  const { camera } = useThree();
  const tl = useRef<gsap.core.Timeline>();

    tl.current = gsap.timeline({ repeat: -1, repeatDelay: 2 });

        { x: 8, y: 4, z: 8 },
        { x: 5, y: 2.5, z: 5, duration: 4, ease: 'power2.inOut' }
      )
      .to(camera.position,

// /components/3d/PerformanceMonitor.tsx
'use client';
import { useFrame } from '@react-three/fiber';
import { useRef, useState } from 'react';

export type QualityLevel = 'high' | 'medium' | 'low' | 'fallback';

export function usePerformanceMonitor(): QualityLevel {
  const [quality, setQuality] = useState<QualityLevel>('high');
  const frameTimes = useRef<number[]>([]);
  const lastTime = useRef(performance.now());

  useFrame(() => {
    const now = performance.now();
    const delta = now - lastTime.current;
    lastTime.current = now;
    frameTimes.current.push(delta);
    if (frameTimes.current.length > 60) frameTimes.current.shift();

    if (frameTimes.current.length === 60) {
      const avg = frameTimes.current.reduce((a, b) => a + b, 0) / 60;
      const fps = 1000 / avg;
      if (fps < 20) setQuality('fallback');
      else if (fps < 30) setQuality('low');
      else if (fps < 45) setQuality('medium');
      else setQuality('high');
    }
  });

  return quality;
}

'use client';
import { useRef, useState, useEffect } from 'react';
import { Html } from '@react-three/drei';
import gsap from 'gsap';
import { ScrollTrigger } from 'gsap/ScrollTrigger';
import CarModel from './CarModel';

gsap.registerPlugin(ScrollTrigger);

const PART_LABELS = [
  { name: 'ENGINE',       position: [0, 2.5, 0.5] },
  { name: 'BRAKES',       position: [-2, -0.5, 1] },
  { name: 'SUSPENSION',   position: [-2.5, -1, 0] },
  { name: 'TRANSMISSION', position: [0, -1.8, -0.3] },
  { name: 'COOLING',      position: [0, 0.8, 2.2] },
  { name: 'ELECTRICAL',   position: [-1.2, 1.8, 0.5] },
  { name: 'BODY',         position: [0, 1.5, 0] },
  { name: 'INTERIOR',     position: [0, 3, 0.3] },
];

export default function DisassemblyScene() {
  const [explodeProgress, setExplodeProgress] = useState(0);
  const [showLabels, setShowLabels] = useState(false);
  const progressRef = useRef({ value: 0 });

  useEffect(() => {
    const trigger = ScrollTrigger.create({
      trigger: '#disassembly-section',
      start: 'top center',
      end: 'bottom center',
      scrub: 1,
      onUpdate: (self) => {
        const progress = self.progress;
        if (progress <= 0.7) {
          const explodeVal = progress / 0.7;
          gsap.to(progressRef.current, {
            value: explodeVal, duration: 0.3,
            onUpdate: () => setExplodeProgress(progressRef.current.value),
          });
          setShowLabels(explodeVal > 0.5);
        } else if (progress <= 0.85) {
          setShowLabels(true);
        } else {
          const reassembleVal = 1 - ((progress - 0.85) / 0.15);
          gsap.to(progressRef.current, {
            value: reassembleVal, duration: 0.3,
            onUpdate: () => setExplodeProgress(progressRef.current.value),
          });
          setShowLabels(reassembleVal > 0.5);
        }
      },
    });
    return () => trigger.kill();
  }, []);

  return (
    <group>
      <CarModel exploded={explodeProgress > 0} explodeProgress={explodeProgress} />
      {showLabels && PART_LABELS.map((label) => (
        <group key={label.name} position={label.position}>
          <Html center distanceFactor={8}>
            <div className="flex items-center gap-2 whitespace-nowrap">
              <div className="w-8 h-px bg-[#D4A853]/40" />
              <span className="font-mono text-[10px] tracking-[0.2em]
                             text-white/60 bg-black/60 backdrop-blur-sm
                             border border-white/10 px-3 py-1.5 rounded">
                {label.name}
              </span>
            </div>
          </Html>
        </group>
      ))}
    </group>
  );
}

// /components/animations/SectionReveal.tsx
'use client';
import { useRef } from 'react';
import gsap from 'gsap';
import { useGSAP } from '@gsap/react';
import { ScrollTrigger } from 'gsap/ScrollTrigger';
gsap.registerPlugin(ScrollTrigger);

export default function SectionReveal({ children, className, delay = 0, direction = 'up' }) {
  const ref = useRef(null);
  useGSAP(() => {
    const fromVars = { opacity: 0 };
    if (direction === 'up') fromVars.y = 60;
    if (direction === 'down') fromVars.y = -60;
    if (direction === 'left') fromVars.x = 60;
    if (direction === 'right') fromVars.x = -60;

    gsap.fromTo(ref.current, fromVars, {
      opacity: 1, x: 0, y: 0,
      duration: 1, delay, ease: 'power3.out',
      scrollTrigger: {
        trigger: ref.current,
        start: 'top 85%',
        toggleActions: 'play none none reverse',
      },
    });
  }, { scope: ref });

  return <div ref={ref} className={className}>{children}</div>;
}

// /components/animations/PageLoader.tsx
'use client';
import { useEffect, useRef } from 'react';
import gsap from 'gsap';

export default function PageLoader({ onComplete }) {
  const loaderRef = useRef(null);
  const ringRef = useRef(null);
  const logoRef = useRef(null);

  useEffect(() => {
    const tl = gsap.timeline({
      onComplete: () => {
        gsap.to(loaderRef.current, { opacity: 0, duration: 0.5, onComplete });
      },
    });
    tl.to(ringRef.current, { rotation: 360, duration: 1.2, ease: 'power2.inOut' })
      .fromTo(logoRef.current,
        { opacity: 0, y: 10 },
        { opacity: 1, y: 0, duration: 0.6, ease: 'power2.out' }, '-=0.3'
      )
      .to({}, { duration: 0.3 });
  }, [onComplete]);

  return (
    <div ref={loaderRef} className="fixed inset-0 z-[100] bg-[#0A0A0B]
         flex items-center justify-center">
      <div className="flex flex-col items-center gap-6">
        <div ref={ringRef} className="relative w-20 h-20">
          <div className="absolute inset-0 rounded-full border border-white/10" />
          <div className="absolute inset-0 rounded-full border-t-2 border-[#D4A853]" />
          <div className="absolute inset-3 rounded-full border border-white/10" />

        </div>
        <div ref={logoRef} className="opacity-0 text-center">
          <p className="font-display text-lg font-bold tracking-[0.3em] text-white">

}

  Cinematic video/render fallback
  Simplified interactions

  Static/high-quality animated fallback
  No 3D rendering
  Pre-rendered images

2. Detect device performance (GPU tier)
3. If WebGL not supported → show static image
4. If low FPS detected → switch to lower LOD
5. If very low FPS → switch to video/image fallback

NEVER allow 3D to make the website unusably slow.

SECTION 35: ASSET OPTIMIZATION PIPELINE
================================================================================

3D MODEL OPTIMIZATION:
──────────────────────
1. Export from source as GLB
2. Compress with Draco: gltf-transform draco input.glb output.glb
3. Reduce polygons if > 80k
4. Compress textures to 2K max
5. Create LOD variants (LOD0, LOD1, LOD2)
6. Verify in https://gltf-viewer.donmccurdy.com/
7. Target: < 5MB per model

IMAGE OPTIMIZATION:
───────────────────
1. Export from source at 2x display size
2. Compress with Squoosh (https://squoosh.app/)
3. Convert to WebP (quality 80-85)
4. Create AVIF variant for modern browsers
5. Create responsive variants (1x, 2x)
6. Add width/height attributes to prevent CLS

TEXTURE OPTIMIZATION:
─────────────────────
1. Resize to 1K or 2K (no need for 4K on web)
2. Compress with KTX2/Basis for GPU-friendly format
3. Use https://github.com/KhronosGroup/KTX-Software

FONT OPTIMIZATION:
──────────────────
1. Use next/font for automatic optimization
2. Subset fonts to Latin characters only
3. Use font-display: swap
4. Preload critical fonts
